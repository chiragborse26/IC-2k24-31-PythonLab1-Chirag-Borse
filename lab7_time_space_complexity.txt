def find_max(arr):
    max_val = arr[0]
    for i in range(1, len(arr)):
        if arr[i] > max_val:
            max_val = arr[i]
    return max_val

Time complexity:  O(n)
Space complexity: O(1)
Justify: Single loop running (n - 1) iterations with constant auxiliary memory.

def has_duplicate(arr):
    for i in range(len(arr)):
        for j in range(i + 1, len(arr)):
            if arr[i] == arr[j]:
                return True
    return False

Time complexity:  O(n^2)
Space complexity: O(1)
Justify: Two nested loops compare each pair of elements (approx. n*(n-1)/2 iterations) with no extra data structures.

def binary_search(arr, target):
    low, high = 0, len(arr) - 1
    while low <= high:
        mid = (low + high) // 2
        if arr[mid] == target:
            return mid
        elif arr[mid] < target:
            low = mid + 1
        else:
            high = mid - 1
    return -1

Time complexity:  O(log n)
Space complexity: O(1)
Justify: The search space is halved in each iteration using iterative pointer updates.

def matrix_multiply(a, b):
    n = len(a)
    result = [[0] * n for _ in range(n)]
    for i in range(n):
        for j in range(n):
            for k in range(n):
                result[i][j] += a[i][k] * b[k][j]
    return result

Time complexity:  O(n^3)
Space complexity: O(n^2)
Justify: Three nested loops each iterate n times to compute dot products, storing an n x n result matrix.

def to_sparse(matrix):
    triples = []
    for r in range(len(matrix)):
        for c in range(len(matrix[0])):
            if matrix[r][c] != 0:
                triples.append((r, c, matrix[r][c]))
    return triples

Time complexity (in terms of matrix dimensions m and n): O(m * n)
Space complexity (in terms of k non-zero elements):      O(k)
Justify: Iterates over all m * n cells and stores tuples only for the k non-zero entries.

def reverse_in_place(arr):
    left, right = 0, len(arr) - 1
    while left < right:
        arr[left], arr[right] = arr[right], arr[left]
        left += 1
        right -= 1
    return arr

Time complexity:  O(n)
Space complexity: O(1)
Justify: Performs n/2 swaps directly in the input array using two pointers without extra space.

Space complexity: O(n)

def count_pairs_with_sum(arr, target):
    seen = set()
    count = 0
    for num in arr:
        if target - num in seen:
            count += 1
        seen.add(num)
    return count

Time complexity:  O(n)
Space complexity: O(n)
Justify: Single pass through array with O(1) average hash set lookups and insertions, storing up to n elements.

def print_all_subsets(arr):
    n = len(arr)
    for i in range(2 ** n):
        subset = []
        for j in range(n):
            if i & (1 << j):
                subset.append(arr[j])
        print(subset)

Time complexity:  O(n * 2^n)
Space complexity: O(n)
Justify: The outer loop runs 2^n times and the inner bitwise check runs n times per subset.

def merge_sorted(a, b):
    result = []
    i = j = 0
    while i < len(a) and j < len(b):
        if a[i] <= b[j]:
            result.append(a[i])
            i += 1
        else:
            result.append(b[j])
            j += 1
    result.extend(a[i:])
    result.extend(b[j:])
    return result

Time complexity:  O(len(a) + len(b))
Space complexity: O(len(a) + len(b))
Justify: Iterates simultaneously through both arrays in linear time and accumulates elements into a merged list.

    for row in matrix:
        for val in row:
            flat.append(val)
    return flat

Time complexity (in terms of m rows and n columns): O(m * n)
Space complexity:                                    O(m * n)
Justify: Iterates through each of the m * n items and stores them in a single flattened 1D list.

Space complexity: O(exp)
Justify: Makes 'exp' sequential recursive calls until reaching the base case, using exp call stack frames.

def fast_power(base, exp):
    if exp == 0:
        return 1
    half = fast_power(base, exp // 2)
    if exp % 2 == 0:
        return half * half
    return half * half * base

Time complexity:  O(log exp)
Space complexity: O(log exp)
Justify: Exponent is divided by 2 at each recursive level, requiring logarithmic iterations and stack depth.

def build_frequency_map(arr):
    freq = {}
    for val in arr:
        freq[val] = freq.get(val, 0) + 1
    return freq

Space complexity: O(u) [where u <= n is count of unique elements]
Justify: Iterates over n elements with average O(1) hash map lookups and updates, storing unique elements.
